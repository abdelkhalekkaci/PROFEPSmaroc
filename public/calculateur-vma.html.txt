    <script>
        // Menu mobile
        const navbar = document.getElementById("navbar");
        const navToggle = document.getElementById("navToggle");
        const navMenu = document.getElementById("navMenu");
        window.addEventListener("scroll", () => { navbar.classList.toggle("scrolled", window.scrollY > 40); });
        if (navToggle && navMenu) {
            navToggle.addEventListener("click", () => {
                navMenu.classList.toggle("active");
            });
        }

        // Onglets de test
        let currentTest = "direct";
        const tabBtns = document.querySelectorAll(".test-tab-btn");
        const paneDirect = document.getElementById("test-pane-direct");
        const paneCooper = document.getElementById("test-pane-demi-cooper");
        const paneLuc = document.getElementById("test-pane-luc-leger");

        tabBtns.forEach(btn => {
            btn.addEventListener("click", () => {
                tabBtns.forEach(b => b.classList.remove("active"));
                btn.classList.add("active");
                currentTest = btn.dataset.test;

                paneDirect.style.display = currentTest === "direct" ? "block" : "none";
                paneCooper.style.display = currentTest === "demi-cooper" ? "block" : "none";
                paneLuc.style.display = currentTest === "luc-leger" ? "block" : "none";
            });
        });

        // Calcul
        const btnCalc = document.getElementById("btnCalculate");
        btnCalc.addEventListener("click", triggerCalculate);

        function triggerCalculate() {
            let vma = 0;
            if (currentTest === "direct") {
                vma = parseFloat(document.getElementById("directVma").value) || 0;
            } else if (currentTest === "demi-cooper") {
                const dist = parseFloat(document.getElementById("cooperDist").value) || 0;
                vma = dist / 100;
            } else if (currentTest === "luc-leger") {
                const palier = parseFloat(document.getElementById("lucPalier").value) || 8;
                vma = 8 + (palier * 0.5);
            }

            if (vma <= 0 || isNaN(vma)) {
                alert("Veuillez saisir une valeur valide.");
                return;
            }

            const trackDist = parseFloat(document.getElementById("trackLength").value) || 250;
            const targetIntensity = parseFloat(document.getElementById("targetIntensity").value) || 90;

            // Affichage résultats
            document.getElementById("results-placeholder").style.display = "none";
            const resContainer = document.getElementById("results-container");
            resContainer.style.display = "block";

            document.getElementById("resVma").textContent = vma.toFixed(1) + " km/h";
            document.getElementById("resVo2").textContent = (vma * 3.5).toFixed(1) + " ml/kg/min";
            document.getElementById("resTrackDistLabel").textContent = "Piste de " + trackDist + "m";

            const targetSpeedKmh = vma * (targetIntensity / 100);
            const targetSpeedMs = targetSpeedKmh / 3.6;
            document.getElementById("resSpeedTarget").textContent = targetSpeedKmh.toFixed(1) + " km/h (" + targetIntensity + "%)";

            const lapSec = trackDist / targetSpeedMs;
            const lapMin = Math.floor(lapSec / 60);
            const lapRestSec = Math.round(lapSec % 60);
            document.getElementById("resLapTarget").textContent = (lapMin > 0 ? lapMin + "m " : "") + (lapRestSec < 10 ? "0" : "") + lapRestSec + "s";

            // Tableau des allures
            const intensities = [
                { pct: 60, name: "Récupération", color: "#64748b" },
                { pct: 70, name: "Aérobie Basse", color: "#0284c7" },
                { pct: 75, name: "Endurance", color: "#0ea5e9" },
                { pct: 80, name: "Endurance Active", color: "#10b981" },
                { pct: 85, name: "Bac EPS (Long)", color: "#059669" },
                { pct: 90, name: "Allure 3x500m", color: "#eab308" },
                { pct: 95, name: "Seuil Lactique", color: "#f97316" },
                { pct: 100, name: "VMA (100%)", color: "#ef4444" },
                { pct: 105, name: "Sur-VMA (Court)", color: "#dc2626" },
                { pct: 110, name: "Sprint Intermittent", color: "#991b1b" }
            ];

            const tbody = document.getElementById("alluresTbody");
            tbody.innerHTML = "";

            intensities.forEach(item => {
                const spdKmh = vma * (item.pct / 100);
                const spdMs = spdKmh / 3.6;
                const secKm = 3600 / spdKmh;
                const pMin = Math.floor(secKm / 60);
                const pSec = Math.round(secKm % 60);
                const paceStr = pMin + ":" + (pSec < 10 ? "0" : "") + pSec;

                const time100 = (100 / spdMs).toFixed(1) + "s";
                const tourSec = trackDist / spdMs;
                const tMin = Math.floor(tourSec / 60);
                const tSec = Math.round(tourSec % 60);
                const tourStr = (tMin > 0 ? tMin + "m " : "") + (tSec < 10 ? "0" : "") + tSec + "s";
                const dist10 = Math.round(spdMs * 600) + "m";

                const isSel = Math.abs(targetIntensity - item.pct) <= 2;
                const tr = document.createElement("tr");
                if (isSel) tr.className = "selected-row";
                tr.innerHTML = `<td><strong>${item.pct}%</strong></td>
                                <td><span class="zone-pill" style="background:${item.color}18; color:${item.color};">${item.name}</span></td>
                                <td><strong>${spdKmh.toFixed(1)}</strong> km/h</td>
                                <td>${paceStr}</td>
                                <td>${time100}</td>
                                <td><strong>${tourStr}</strong></td>
                                <td>${dist10}</td>`;
                tbody.appendChild(tr);
            });
        }
